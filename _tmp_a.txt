206:function isHttp(u) {
207:  return typeof u === 'string' && (u.indexOf('http://') === 0 || u.indexOf('https://') === 0);
208:}
209:function isWeb(u) {
210:  return isHttp(u) || (typeof u === 'string' && u.indexOf('mailto:') === 0);
211:}
212:function detailLogo(name, url) {
213:  if (isHttp(url)) return '<div class="detail-logo"><img src="' + esc(url) + '" alt="" loading="lazy"></div>';
214:  return '<div class="detail-logo">' + esc(initials(name)) + '</div>';
215:}
216:/* Pick ONE best official external link for an internship opportunity. */
217:function pickOppLink(o) {
218:  const cand = [
219:    [o.application_url, 'View Official Application'],
220:    [o.source_url, 'View Official Application'],
221:    [o.company_careers_url, 'View Official Application'],
222:    [o.company_official_website, 'Visit Official Company'],
223:    [o.company_website, 'Visit Official Company']
224:  ];
225:  for (const [u, label] of cand) if (isHttp(u)) return { url: u, label };
226:  return null;
227:}
228:/* Pick ONE best official external link for a company. */
229:function pickCompanyLink(c) {
230:  const cand = [
231:    [c.careers_url, 'View Official Application'],
232:    [c.official_website, 'Visit Official Company'],
233:    [c.website, 'Visit Official Company'],
234:    [c.source_url, 'Visit Official Company']
235:  ];
236:  for (const [u, label] of cand) if (isHttp(u)) return { url: u, label };
237:  if (c.contact_info) {
238:    const m = c.contact_info.match(/[\w.+-]+@[\w-]+(?:\.[\w-]+)+/i);
239:    if (m && m[0]) return { url: 'mailto:' + m[0], label: 'Contact Company' };
240:  }
241:  return null;
242:}
243:/* Single, clean detail modal template used everywhere. */
244:function detailSheet(o) {
245:  const progs = (o.programs || []);
246:  const progTags = progs.map((p) => '<span class="tag">' + esc(p) + '</span>').join('');
247:  const addressHtml = o.address
248:    ? '<div class="detail-address"><span class="d-ico">&#128205;</span><span>' + esc(o.address) + '</span></div>'
249:    : '';
250:  const cta = (o.href && isWeb(o.href))
251:    ? '<a class="btn btn-primary detail-cta" href="' + esc(o.href) + '" target="_blank" rel="noopener noreferrer">' + esc(o.label || 'View Official Application') + ' &#8599;</a>'
252:    : '<p class="detail-note">No official application link is available for this listing.</p>';
253:  return '<div class="detail-sheet">' +
254:    '<div class="detail-head">' + detailLogo(o.name, o.logo) + '<h3 class="detail-name">' + esc(o.name) + '</h3></div>' +
255:    (o.title ? '<div class="detail-role">' + esc(o.title) + '</div>' : '') +
256:    (o.desc ? '<p class="detail-desc">' + esc(o.desc) + '</p>' : '') +
257:    addressHtml +
258:    (progs.length ? '<div class="detail-progs"><span class="detail-label">Matched Programs</span><div class="tag-row">' + progTags + '</div></div>' : '') +
259:    cta +
260:  '</div>';
