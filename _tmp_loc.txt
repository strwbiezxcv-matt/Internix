106:function findMunicipality(token) {
107:  const n = norm(token);
108:  if (!n) return null;
109:  if (BULACAN_BY_NORM.has(n)) return BULACAN_BY_NORM.get(n);
110:  if (METRO_BY_NORM.has(n)) return METRO_BY_NORM.get(n);
111:  const alias = ALIAS_NORM[n];
112:  if (alias) return alias;
113:  // Any alias whose normalized form is part of the token (e.g. "Santa Maria, Bulacan")
114:  for (const k in ALIAS_NORM) {
115:    if (n === k || n.startsWith(k + ' ') || n.includes(' ' + k + ' ') || n.endsWith(' ' + k)) return ALIAS_NORM[k];
116:  }
117:  return null;
118:}
119:/**
120: * Parse a free-text/structured location into a normalised object.
121: * Expects a string such as "Santa Maria, Bulacan", "Makati, Metro Manila",
122: * "Malolos", "Quezon City", or an object with municipality/province fields.
123: * Returns { municipality, city, province, region, label } or null.
124: */
125:function parseLocation(value) {
126:  if (!value) return null;
127:  let municipality = null;
128:  let city = null;
129:  let province = null;
130:  let raw = '';
131:
132:  if (typeof value === 'object') {
133:    municipality = value.municipality || value.city || null;
134:    city = value.city || value.municipality || null;
135:    province = value.province || value.region || null;
136:    raw = value.raw || [municipality, province].filter(Boolean).join(', ');
137:    if (value.region) province = value.region;
138:  } else {
139:    raw = String(value).trim();
140:    const parts = raw.split(',').map((p) => p.trim()).filter(Boolean);
141:    let provinceToken = null;
142:    for (const p of parts) {
143:      const n = norm(p);
144:      if (n === 'bulacan') provinceToken = 'Bulacan';
145:      else if (n === 'metro manila' || n === 'metromanila') provinceToken = 'Metro Manila';
146:    }
147:    if (provinceToken) {
148:      province = provinceToken;
149:      for (const p of parts) {
150:        const m = findMunicipality(p);
151:        if (m) { municipality = m; break; }
152:      }
153:      if (!municipality) municipality = parts[0];
154:    } else {
155:      // Single token like "Malolos" or "Santa Maria".
156:      const m = findMunicipality(raw);
157:      if (m) {
158:        municipality = m;
159:        province = BULACAN_SET.has(norm(m)) ? 'Bulacan' : (METRO_SET.has(norm(m)) ? 'Metro Manila' : null);
160:      } else {
161:        municipality = parts[0];
162:      }
163:    }
164:  }
165:
166:  municipality = municipality && String(municipality).trim();
167:  city = (city || municipality);
168:  if (city) city = String(city).trim();
169:
170:  if (!province) {
171:    if (municipality && BULACAN_SET.has(norm(municipality))) province = 'Bulacan';
172:    else if (municipality && METRO_SET.has(norm(municipality))) province = 'Metro Manila';
173:  }
174:
175:  const region = (province === 'Bulacan') ? REGION.BULACAN
176:    : (province === 'Metro Manila' ? REGION.METRO : null);
177:
178:  const label = municipality
179:    ? (region ? `${municipality}, ${region}` : municipality)
180:    : (raw || (province || ''));
181:
182:  return {
183:    municipality: municipality || null,
184:    city: city || null,
185:    province: province || null,
186:    region,
187:    label
188:  };
189:}
