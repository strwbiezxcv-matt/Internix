266:/**
267: * Resolve a program key (a program code, a full program name, or a
268: * specialization name) into { program_id, code, name, specialization }.
269: * The BINDTECH specialization names (e.g. "Computer Technology") resolve to
270: * their parent program (BINDTECH) with the specialization attached.
271: */
272:function resolveProgramKey(key) {
273:  if (!key) return null;
274:  const s = String(key).trim();
275:  const byCode = db.get('SELECT id, code, name FROM programs WHERE code = ?', s);
276:  if (byCode) return { program_id: byCode.id, code: byCode.code, name: byCode.name, specialization: null };
277:  const byName = db.get('SELECT id, code, name FROM programs WHERE lower(name) = lower(?)', s);
278:  if (byName) return { program_id: byName.id, code: byName.code, name: byName.name, specialization: null };
279:  const spec = db.get(
280:    `SELECT s.name AS spec, p.id AS program_id, p.code, p.name
281:       FROM specializations s JOIN programs p ON p.id = s.program_id
282:      WHERE lower(s.name) = lower(?)`, s
283:  );
284:  if (spec) return { program_id: spec.program_id, code: spec.code, name: spec.name, specialization: spec.spec };
285:
286:  // Also check specializations by name (for BINDTECH specializations and others)
287:  const specName = db.get(
288:    `SELECT s.name AS spec, p.id AS program_id, p.code, p.name
289:       FROM specializations s JOIN programs p ON p.id = s.program_id
290:      WHERE lower(s.name) = lower(?)`, s
291:  );
292:  if (specName) return { program_id: specName.program_id, code: specName.code, name: specName.name, specialization: specName.spec };
293:
294:  return null;
295:}
