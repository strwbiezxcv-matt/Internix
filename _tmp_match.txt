105:/* Group ranked matches by company so the same company never repeats as a
106:   bare duplicate row: each company appears once with its openings listed
107:   underneath (the best opening sets the headline score). */
108:function dedupeByCompany(ranked) {
109:  const groups = [];
110:  const byId = new Map();
111:  for (const r of (ranked || [])) {
112:    const cid = r && r.opportunity ? r.opportunity.company_id : null;
113:    if (cid === null || cid === undefined || !byId.has(cid)) {
114:      const g = { company_id: cid, best: r, openings: [r] };
115:      byId.set(cid, g);
116:      groups.push(g);
117:    } else { byId.get(cid).openings.push(r); }
118:  }
119:  groups.sort((a, b) => (b.best.match.score - a.best.match.score) ||
120:    String(a.best.opportunity.company_name || '').localeCompare(String(b.best.opportunity.company_name || '')));
121:  return groups;
122:}

124:/* Company-level program gate: a company is listed under selected programs
125:   only when its own relevant_program_codes contain one (or a related one). */
126:function companyMatchesPrograms(company, selectedPrograms) {
127:  const sel = (selectedPrograms || []).filter(Boolean);
128:  if (!sel.length) return { ok: true, tier: 'none' };
129:  const codes = (company && company.relevant_program_codes) || [];
130:  if (codes.some((c) => sel.includes(c))) return { ok: true, tier: 'exact' };
131:  const relHit = sel.some((sc) => programMap.relatedFor(sc).some((r) => codes.includes(r)));
132:  if (relHit) return { ok: true, tier: 'related' };
133:  if (!codes.length) return { ok: true, tier: 'open' };
134:  return { ok: false, tier: 'mismatch' };
135:}

138:function computeMatch(profile, opp) {
139:  profile = profile || {};
140:  opp = opp || {};
141:  const selected = normalizeSelected(profile);
142:  const oppPrograms = opp.programs || [];
143:
144:  const prog = programMap.programMatch(selected, oppPrograms);
145:  const lc = locations.locationScore(profile.location, opp._loc);
146:  const role = roleScore(selected, opp.field);
147:  const skills = skillsScore(selected, opp.skills);
148:
149:  const raw = prog.score * WEIGHTS.program + lc * WEIGHTS.location +
150:              role * WEIGHTS.role + skills * WEIGHTS.skills;
151:  const scored = clamp(raw * 100);
152:  const explanation = buildExplanation({ programs: selected, location: profile.location || null }, opp, prog, lc, role, skills);
153:  const checks = buildChecks({ programs: selected, location: profile.location || null }, opp, prog, lc, role, skills);
154:
155:  return {
156:    score: scored,
157:    program: { score: prog.score, tier: prog.tier, matches: prog.matches },
158:    location: lc,
159:    role: role,
160:    skills: skills,
161:    weights: WEIGHTS,
162:    explanation: explanation,
163:    checks: checks,
164:    breakdown: {
165:      program: Math.round(prog.score * WEIGHTS.program * 100),
166:      location: Math.round(lc * WEIGHTS.location * 100),
167:      role: Math.round(role * WEIGHTS.role * 100),
168:      skills: Math.round(skills * WEIGHTS.skills * 100)
169:    }
170:  };
171:}
172:/**
173: * Build a transparent, criteria-based checklist explaining WHY an opportunity
174: * matched. Each check carries its own state so the UI can render ✓ / △ / ✗.
175: * Nothing here is random - every value comes from the real program mapping,
176: * resolved location, and role/skills comparison done above.
177: */
178:function buildChecks(profile, opp, prog, lc, role, skills) {
179:  const selNames = (profile.programs || []).map((c) => programMap.displayName(c));
180:  const oppLoc = opp._loc;
181:  const locLabel = oppLoc && oppLoc.label ? oppLoc.label : (opp.location || 'unlisted location');
182:  const roleField = opp.field || 'not specified';
183:
184:  let programLabel;
185:  let programOk;
186:  if (prog.tier === 'exact') {
187:    const matched = (prog.matches || []).map((c) => programMap.displayName(c));
188:    programLabel = matched.length > 1
189:      ? 'Program: matches ' + matched.join(' & ')
190:      : 'Program: matches ' + (matched[0] || selNames[0] || 'your program');
191:    programOk = true;
192:  } else if (prog.tier === 'related') {
193:    programLabel = 'Program: closely related to your selection';
194:    programOk = true;
195:  } else if (prog.tier === 'open') {
196:    programLabel = 'Program: open to all programs';
197:    programOk = null;
198:  } else {
199:    programLabel = 'Program: ' + (selNames.length ? selNames.join(' & ') : 'not restricted');
200:    programOk = false;
201:  }
202:
203:  const locOk = lc >= 0.9 ? true : (lc >= 0.5 ? null : false);
204:  const roleOk = role >= 0.7 ? true : (role >= 0.4 ? null : false);
205:  const skillOk = skills >= 0.6 ? true : (skills >= 0.35 ? null : false);
206:
207:  return [
208:    { key: 'program', label: programLabel, ok: programOk },
209:    { key: 'location', label: 'Location: ' + locLabel, ok: locOk },
210:    { key: 'role', label: 'Role relevance: ' + roleField, ok: roleOk },
211:    { key: 'skills', label: 'Skills fit: ' + (skills >= 0.6 ? 'strong' : skills >= 0.35 ? 'partial' : 'limited'), ok: skillOk }
212:  ];
213:}
214:function buildExplanation(profile, opp, prog, lc, role, skills) {
215:  const selNames = (profile.programs || []).map((c) => programMap.displayName(c));
216:  const oppLoc = opp._loc || null;
217:  const locName = oppLoc && oppLoc.label ? oppLoc.label : (opp.location || 'unlisted location');
218:
219:  let kind;
220:  let programPart;
221:  if (prog.tier === 'exact') {
222:    const matched = (prog.matches || []).map((c) => programMap.displayName(c));
223:    kind = 'Strong match';
224:    programPart = 'this opportunity explicitly accepts ' +
225:      (matched.length > 1 ? 'multiple of your selected programs' : (matched[0] || 'your program')) + '.';
226:  } else if (prog.tier === 'related') {
227:    kind = 'Related opportunity';
228:    programPart = 'your program is closely related to one of the accepted programs.';
229:  } else if (prog.tier === 'open') {
230:    kind = 'Open opportunity';
231:    programPart = 'this opportunity is open to all programs.';
232:  } else {
233:    kind = 'Potential mismatch';
234:    programPart = 'this opportunity does not list your program.';
235:  }
236:
237:  const locationBits = [];
238:  const sel = profile.location;
239:  if (sel) {
240:    if (sel.municipality) {
241:      if (oppLoc && oppLoc.municipality && String(oppLoc.municipality).toLowerCase() === String(sel.municipality).toLowerCase()) {
242:        locationBits.push('located exactly in ' + sel.municipality);
243:      } else if (oppLoc && oppLoc.province && sel.province && String(oppLoc.province).toLowerCase() === String(sel.province).toLowerCase()) {
244:        locationBits.push('located in ' + (oppLoc.label || oppLoc.province) + ' within your preferred province');
245:      } else {
246:        locationBits.push('located outside your preferred area (' + locName + ')');
247:      }
248:    } else if (sel.province) {
249:      if (oppLoc && oppLoc.province && String(oppLoc.province).toLowerCase() === String(sel.province).toLowerCase()) {
250:        locationBits.push('located in ' + (oppLoc.label || oppLoc.province));
251:      } else {
252:        locationBits.push('located in ' + (oppLoc.label || oppLoc.province || 'another area'));
253:      }
254:    }
255:  }
256:
257:  const skillBit = skills >= 0.65
258:    ? ' It also calls for skills you would develop in ' + (selNames.length ? selNames[0] : 'your program') + '.'
259:    : '';
260:
261:  const text = kind + ' because ' + programPart +
262:    (locationBits.length ? ' It is ' + locationBits.join(', ') + '.' : '') +
263:    ' The role aligns well with ' + (selNames.length ? selNames.join(' and ') : 'your program') + '.' +
264:    skillBit;
265:
266:  return text;
267:}
